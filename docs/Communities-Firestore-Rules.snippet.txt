// Required for /admin/communities — admins must be able to delete communities.
// Registration still creates/updates communities for authenticated users.

    match /communities/{communityId} {
      // Community picker, checkout, registration
      allow read: if true;

      // Authenticated users may create/update public community fields.
      // Do not store membership arrays such as userIds on these documents —
      // the homepage used to download them with every community list load.
      allow create, update: if isAuthenticated();

      // Community admin page — remove obsolete communities
      allow delete: if isAdmin();
    }
