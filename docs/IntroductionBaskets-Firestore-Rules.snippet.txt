// Firestore rules snippet for Introduction baskets.
// Add INSIDE `match /databases/{database}/documents { ... }`, just BEFORE the
// final catch-all deny:  match /{document=**} { allow read, write: if false; }
//
// This project already defines isAdmin() (UID allowlist) at the top of the rules,
// so this block just reuses it. Do NOT redefine isAdmin().
//
// Collection: introductionBaskets/{basketId}
// - Public storefront reads require active == true (customers are anonymous).
// - Only admins can write.

match /introductionBaskets/{basketId} {
  allow read: if resource.data.active == true || isAdmin();
  allow write: if isAdmin();
}
